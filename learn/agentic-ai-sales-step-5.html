Worked SQL lab

lab_5.py

import sqlite3

def connect() -> sqlite3.Connection:
    db = sqlite3.connect(":memory:")
    db.execute("""CREATE TABLE leads (
        tenant_id TEXT NOT NULL,
        external_id TEXT NOT NULL,
        company TEXT NOT NULL,
        PRIMARY KEY (tenant_id, external_id)
    )""")
    return db

def save_lead(db, tenant_id: str, external_id: str, company: str) -> None:
    if not all(value.strip() for value in (tenant_id, external_id, company)):
        raise ValueError("Lead fields must be non-empty")
    with db:
        db.execute(
            "INSERT INTO leads VALUES (?, ?, ?) "
            "ON CONFLICT(tenant_id, external_id) DO UPDATE SET company=excluded.company",
            (tenant_id, external_id, company),
        )

def lookup(db, tenant_id: str, external_id: str):
    return db.execute(
        "SELECT company FROM leads WHERE tenant_id=? AND external_id=?",
        (tenant_id, external_id),
    ).fetchone()

if __name__ == "__main__":
    db = connect()
    save_lead(db, "tenant-a", "crm-1", "Example Tools")
    save_lead(db, "tenant-a", "crm-1", "Example Tools Updated")
    assert lookup(db, "tenant-a", "crm-1") == ("Example Tools Updated",)
    assert lookup(db, "tenant-b", "crm-1") is None
    assert db.execute("SELECT COUNT(*) FROM leads").fetchone()[0] == 1
    print("Deduplication and tenant-scoped lookup passed")
